68 void Init(float sr, bool quantize_semitones = false)
69 {
70 force_recalc_ = false;
71 sr_ = sr;
72 mod_freq_ = 5.0f;
73 for(uint8_t i = 0; i < 2; i++)
74 {
75 gain_[i] = 0.0f;
76 d_[i].Init();
77 phs_[i].Init(sr, 50, i == 0 ? 0 : PI_F);
78 }
79 del_size_ = kShiftDelaySize;
80 SetDelSize(del_size_);
81 fun_ = 0.0f;
82 quantize_semitones_ = quantize_semitones;
83 }

87 float Process(float in)
88 {
89 float val, fade1, fade2;
90 // First Process delay mod/crossfade
91 fade1 = phs_[0].Process();
92 fade2 = phs_[1].Process();
93 bool recalc_a_fun = ((transpose_ >= 0.f) && (prev_phs_a_ > fade1))
94 || ((transpose_ < 0.f) && prev_phs_a_ < fade1);
95 bool recalc_b_fun = ((transpose_ >= 0.f) && (prev_phs_b_ > fade2))
96 || ((transpose_ < 0.f) && prev_phs_b_ < fade2);
97 // Randomly trigger mod recalc when tranpose is 0 so that the "fun"
98 // parameter still does something when not transposing
99 bool rand_retrig = (transpose_ >= -0.25f && transpose_ < 0.25f);
100 if(rand_retrig && fun_ > 0.f)
101 if(myrand() % 65536 < 4)
102 recalc_a_fun = recalc_b_fun = true;
103 if(recalc_a_fun)
104 {
105 mod_a_amt_ = fun_ * ((float)(myrand() % 255) / 255.0f)
106 * (del_size_ * 0.5f);
107 mod_coeff_[0]
108 = 0.0002f + (((float)(myrand() % 255) / 255.0f) * 0.001f);
109 }
110 if(recalc_b_fun)
111 {
112 mod_b_amt_ = fun_ * ((float)(myrand() % 255) / 255.0f)
113 * (del_size_ * 0.5f);
114 mod_coeff_[1]
115 = 0.0002f + (((float)(myrand() % 255) / 255.0f) * 0.001f);
116 }
117 slewed_mod_[0] += mod_coeff_[0] * (mod_a_amt_ - slewed_mod_[0]);
118 slewed_mod_[1] += mod_coeff_[1] * (mod_b_amt_ - slewed_mod_[1]);
119 prev_phs_a_ = fade1;
120 prev_phs_b_ = fade2;
121 fade1 = 1.0f - fade1;
122 fade2 = 1.0f - fade2;
123 mod_[0] = fade1 * (del_size_ - 1);
124 mod_[1] = fade2 * (del_size_ - 1);
125#ifdef USE_ARM_DSP
126 gain_[0] = arm_sin_f32(fade1 * (float)M_PI);
127 gain_[1] = arm_sin_f32(fade2 * (float)M_PI);
128#else
129 gain_[0] = sinf(fade1 * PI_F);
130 gain_[1] = sinf(fade2 * PI_F);
131#endif
132
133 // Handle Delay Writing
134 d_[0].Write(in);
135 d_[1].Write(in);
136 // Modulate Delay Lines
137 // d_[0].SetDelay(mod_[0] + mod_a_amt_);
138 // d_[1].SetDelay(mod_[1] + mod_b_amt_);
139 d_[0].SetDelay(mod_[0] + slewed_mod_[0]);
140 d_[1].SetDelay(mod_[1] + slewed_mod_[1]);
141 val = 0.0f;
142 val += (d_[0].Read() * gain_[0]);
143 val += (d_[1].Read() * gain_[1]);
144 return val;
145 }

149 void SetTransposition(const float &transpose)
150 {
151 float ratio;
152 if(transpose_ != transpose || force_recalc_)
153 {
154 transpose_ = quantize_semitones_ ? (int32_t)transpose : transpose;
155 ratio = pow(2.f, transpose_ / 12.f);
156 mod_freq_ = ((ratio - 1.0f) * sr_) / del_size_;
157 phs_[0].SetFreq(mod_freq_);
158 phs_[1].SetFreq(mod_freq_);
159 if(force_recalc_)
160 {
161 force_recalc_ = false;
162 }
163 }
164 }
165
168 void SetDelSize(uint32_t size)
169 {
170 del_size_ = size < kShiftDelaySize ? size : kShiftDelaySize;
171 force_recalc_ = true;
172 SetTransposition(transpose_);
173 }

177 inline void SetFun(float f) { fun_ = f; }

185 float pitch_shift_, mod_freq_;
